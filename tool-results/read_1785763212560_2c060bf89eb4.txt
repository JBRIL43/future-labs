    30→    30→
    31→    31→function easeOutCubic(t: number): number {
    32→    32→  return 1 - Math.pow(1 - Math.min(1, Math.max(0, t)), 3)
    33→    33→}
    34→    34→
    35→    35→function easeOutExpo(t: number): number {
    36→    36→  return t >= 1 ? 1 : 1 - Math.pow(2, -10 * Math.min(1, Math.max(0, t)))
    37→    37→}

    39→    39→function getModulePos(index: number, time: number, radius: number) {
    40→    40→  const baseAngle = (index / MODULE_COUNT) * Math.PI * 2
    41→    41→  const speed = 0.12 + index * 0.015
    42→    42→  const angle = baseAngle + time * speed
    43→    43→  const yBase = Math.sin(index * 1.7) * 0.25
    44→    44→  const yFloat = Math.sin(time * 0.4 + index * 0.8) * 0.15
    45→    45→  return new THREE.Vector3(
    46→    46→    Math.cos(angle) * radius,
    47→    47→    yBase + yFloat,
    48→    48→    Math.sin(angle) * radius
    49→    49→  )
    50→    50→}

   242→   242→function OrbitingModule({ index }: { index: number }) {

   255→   255→    const entryDelay = 1.5 + index * 0.18

   258→   258→
   259→   259→    const pos = getModulePos(index, t, currentRadius)

   265→   265→      meshRef.current.rotation.x = t * 0.4 + index

   270→   270→    if (glowRef.current) {
   271→   271→      glowRef.current.scale.setScalar(scaleRef.current * entryProgress * 1.8)
   272→   272→      ;(glowRef.current.material as THREE.MeshBasicMaterial).opacity = hovered ? 0.12 : 0.04
   273→   273→    }

   294→   294→        <meshBasicMaterial

   296→   296→          transparent

   298→   298→          blending={THREE.AdditiveBlending}
   299→   299→          side={THREE.BackSide}
   300→   300→        />
   301→   301→      </mesh>

   332→   332→const BEAM_POS = new Float32Array(MODULE_COUNT * 6)
   333→   333→const PARTICLE_POS = new Float32Array(MODULE_COUNT * 3 * 3)
   334→   334→let BEAM_ATTR: THREE.BufferAttribute | null = null
   335→   335→let PARTICLE_ATTR: THREE.BufferAttribute | null = null
   336→   336→const LINE_MAT = new THREE.LineBasicMaterial({
   337→   337→  color: '#10b981', transparent: true, opacity: 0, blending: THREE.AdditiveBlending,

   339→   339→const POINT_MAT = new THREE.PointsMaterial({

   343→   343→
   344→   344→function EnergyBeams() {
   345→   345→  const [lineObj] = useState(() => {
   346→   346→    const g = new THREE.BufferGeometry()
   347→   347→    const attr = new THREE.BufferAttribute(BEAM_POS, 3)
   348→   348→    g.setAttribute('position', attr)
   349→   349→    BEAM_ATTR = attr
   350→   350→    return new THREE.LineSegments(g, LINE_MAT)
   351→   351→  })
   352→   352→  const [pointsObj] = useState(() => {
   353→   353→    const g = new THREE.BufferGeometry()
   354→   354→    const attr = new THREE.BufferAttribute(PARTICLE_POS, 3)
   355→   355→    g.setAttribute('position', attr)
   356→   356→    PARTICLE_ATTR = attr
   357→   357→    return new THREE.Points(g, POINT_MAT)
   358→   358→  })
   359→   359→
   360→   360→  useFrame(({ clock }) => {
   361→   361→    const t = clock.getElapsedTime()
   362→   362→    if (!sceneVisible.current) return
   363→   363→
   364→   364→    let anyVisible = false
   365→   365→
   366→   366→    for (let i = 0; i < MODULE_COUNT; i++) {
   367→   367→      const entryDelay = 1.5 + i * 0.18
   368→   368→      const entryProgress = easeOutCubic((t - entryDelay) / 0.9)
   369→   369→      const currentRadius = ORBIT_RADIUS * entryProgress
   370→   370→      const pos = getModulePos(i, t, currentRadius)
   371→   371→
   372→   372→      const idx = i * 6
   373→   373→      BEAM_POS[idx] = 0
   374→   374→      BEAM_POS[idx + 1] = 0
   375→   375→      BEAM_POS[idx + 2] = 0
   376→   376→      BEAM_POS[idx + 3] = pos.x
   377→   377→      BEAM_POS[idx + 4] = pos.y
   378→   378→      BEAM_POS[idx + 5] = pos.z
   379→   379→
   380→   380→      if (entryProgress > 0.1) anyVisible = true
   381→   381→
   382→   382→      for (let p = 0; p < 3; p++) {
   383→   383→        const pIdx = (i * 3 + p) * 3
   384→   384→        const progress = ((t * 0.8 + p * 0.33 + i * 0.17) % 1)
   385→   385→        PARTICLE_POS[pIdx] = pos.x * progress
   386→   386→        PARTICLE_POS[pIdx + 1] = pos.y * progress
   387→   387→        PARTICLE_POS[pIdx + 2] = pos.z * progress
   388→   388→      }
   389→   389→    }
   390→   390→
   391→   391→    if (BEAM_ATTR) BEAM_ATTR.needsUpdate = true
   392→   392→    if (PARTICLE_ATTR) PARTICLE_ATTR.needsUpdate = true
   393→   393→
   394→   394→    LINE_MAT.opacity = anyVisible ? 0.06 + Math.sin(t * 1.5) * 0.02 : 0
   395→   395→    POINT_MAT.opacity = anyVisible ? 0.6 : 0
   396→   396→  })
   397→   397→
   398→   398→  return (
   399→   399→    <group>
   400→   400→      <primitive object={lineObj} />
   401→   401→      <primitive object={pointsObj} />
   402→   402→    </group>
   403→   403→  )
   404→   404→}

   410→   410→
   411→   411→function AmbientParticles() {
   412→   412→  const meshRef = useRef<THREE.InstancedMesh>(null)
   413→   413→  const dummy = useMemo(() => new THREE.Object3D(), [])
   414→   414→
   415→   415→  const count = typeof window !== 'undefined' && window.innerWidth < 768 ? 80 : 160
   416→   416→
   417→   417→  const particles = useMemo(() => {
   418→   418→    return Array.from({ length: count }, () => ({
   419→   419→      position: new THREE.Vector3(
   420→   420→        (Math.random() - 0.5) * 18,
   421→   421→        (Math.random() - 0.5) * 14,
   422→   422→        (Math.random() - 0.5) * 18
   423→   423→      ),
   424→   424→      speed: Math.random() * 0.25 + 0.05,
   425→   425→      offset: Math.random() * Math.PI * 2,
   426→   426→      size: Math.random() * 0.6 + 0.4,
   427→   427→    }))
   428→   428→  }, [count])
   429→   429→
   430→   430→  useFrame(({ clock }) => {
   431→   431→    if (!meshRef.current || !sceneVisible.current) return
   432→   432→    const t = clock.getElapsedTime()
   433→   433→
   434→   434→    for (let i = 0; i < particles.length; i++) {
   435→   435→      const p = particles[i]
   436→   436→      dummy.position.set(
   437→   437→        p.position.x + Math.sin(t * p.speed + p.offset) * 0.5,
   438→   438→        p.position.y + Math.cos(t * p.speed * 0.6 + p.offset) * 0.35,
   439→   439→        p.position.z + Math.sin(t * p.speed * 0.4 + p.offset * 2) * 0.4
   440→   440→      )
   441→   441→      const s = p.size * (0.6 + Math.sin(t * 0.7 + p.offset) * 0.4)
   442→   442→      dummy.scale.setScalar(s)
   443→   443→      dummy.updateMatrix()
   444→   444→      meshRef.current.setMatrixAt(i, dummy.matrix)
   445→   445→    }
   446→   446→    meshRef.current.instanceMatrix.needsUpdate = true

   520→   520→function NetworkNodes() {
   521→   521→  const meshRef = useRef<THREE.InstancedMesh>(null)
   522→   522→  const lineRef = useRef<THREE.LineSegments>(null)
   523→   523→  const matRef = useRef<THREE.PointsMaterial>(null)
   524→   524→  const lineMatRef = useRef<THREE.LineBasicMaterial>(null)

   526→   526→
   527→   527→  const extraNodes = useMemo(() => {
   528→   528→    const nodes: THREE.Vector3[] = []
   529→   529→    for (let i = 0; i < 20; i++) {
   530→   530→      const theta = Math.random() * Math.PI * 2
   531→   531→      const phi = Math.random() * Math.PI
   532→   532→      const r = 4.5 + Math.random() * 3
   533→   533→      nodes.push(new THREE.Vector3(
   534→   534→        r * Math.sin(phi) * Math.cos(theta),
   535→   535→        (Math.random() - 0.5) * 4,
   536→   536→        r * Math.sin(phi) * Math.sin(theta)
   537→   537→      ))
   538→   538→    }
   539→   539→    return nodes
   540→   540→  }, [])
   541→   541→
   542→   542→  // Connection lines between nearby extra nodes
   543→   543→  const lineGeometry = useMemo(() => {
   544→   544→    const geo = new THREE.BufferGeometry()
   545→   545→    const positions: number[] = []
   546→   546→    for (let i = 0; i < extraNodes.length; i++) {
   547→   547→      for (let j = i + 1; j < extraNodes.length; j++) {
   548→   548→        if (extraNodes[i].distanceTo(extraNodes[j]) < 4.5) {
   549→   549→          positions.push(
   550→   550→            extraNodes[i].x, extraNodes[i].y, extraNodes[i].z,
   551→   551→            extraNodes[j].x, extraNodes[j].y, extraNodes[j].z
   552→   552→          )
   553→   553→        }
   554→   554→      }
   555→   555→    }
   556→   556→    geo.setAttribute('position', new THREE.Float32BufferAttribute(positions, 3))
   557→   557→    return geo
   558→   558→  }, [extraNodes])
   559→   559→
   560→   560→  useFrame(({ clock }) => {
   561→   561→    if (!sceneVisible.current) return
   562→   562→    const t = clock.getElapsedTime()
   563→   563→
   564→   564→    // Network nodes appear at scroll progress > 0.3
   565→   565→    const scrollY = window.scrollY
   566→   566→    const vh = window.innerHeight
   567→   567→    const scrollProgress = Math.min(1, scrollY / (vh * 2.5))
   568→   568→    const networkAlpha = Math.max(0, (scrollProgress - 0.2) / 0.5)
   569→   569→
   570→   570→    if (meshRef.current) {
   571→   571→      for (let i = 0; i < extraNodes.length; i++) {
   572→   572→        const p = extraNodes[i]
   573→   573→        dummy.position.set(
   574→   574→          p.x + Math.sin(t * 0.2 + i) * 0.1,
   575→   575→          p.y + Math.cos(t * 0.15 + i * 0.7) * 0.08,
   576→   576→          p.z + Math.sin(t * 0.18 + i * 1.3) * 0.1
   577→   577→        )
   578→   578→        const s = networkAlpha * (0.5 + Math.sin(t * 0.5 + i) * 0.3)
   579→   579→        dummy.scale.setScalar(Math.max(0.01, s))
   580→   580→        dummy.updateMatrix()
   581→   581→        meshRef.current.setMatrixAt(i, dummy.matrix)
   582→   582→      }
   583→   583→      meshRef.current.instanceMatrix.needsUpdate = true
   584→   584→    }
   585→   585→
   586→   586→    if (matRef.current) {
   587→   587→      matRef.current.opacity = networkAlpha * 0.5
   588→   588→    }
   589→   589→    if (lineMatRef.current) {
   590→   590→      lineMatRef.current.opacity = networkAlpha * 0.06
   591→   591→    }
   592→   592→  })
   593→   593→
   594→   594→  return (
   595→   595→    <group>
   596→   596→      <instancedMesh ref={meshRef} args={[undefined, undefined, extraNodes.length]}>
   597→   597→        <sphereGeometry args={[0.04, 8, 8]} />
   598→   598→        <pointsMaterial ref={matRef} color="#06b6d4" size={0.08} transparent opacity={0} sizeAttenuation />
   599→   599→      </instancedMesh>
   600→   600→      <lineSegments geometry={lineGeometry}>
   601→   601→        <lineBasicMaterial
   602→   602→          ref={lineMatRef}
   603→   603→          color="#06b6d4"
   604→   604→          transparent
   605→   605→          opacity={0}
   606→   606→          blending={THREE.AdditiveBlending}
   607→   607→        />
   608→   608→      </lineSegments>
   609→   609→    </group>
   610→   610→  )

   711→   711→   Mouse parallax + scroll-driven zoom
   712→   712→   ═══════════════════════════════════════════ */
   713→   713→
   714→   714→function CameraController() {
   715→   715→  const { camera } = useThree()
   716→   716→  const mouse = useRef({ x: 0, y: 0 })
   717→   717→  const target = useRef(new THREE.Vector3(0, 0, 7.5))
   718→   718→
   719→   719→  useEffect(() => {
   720→   720→    const onMove = (e: MouseEvent) => {
   721→   721→      mouse.current.x = (e.clientX / window.innerWidth - 0.5) * 2
   722→   722→      mouse.current.y = (e.clientY / window.innerHeight - 0.5) * 2
   723→   723→    }
   724→   724→    window.addEventListener('mousemove', onMove, { passive: true })
   725→   725→    return () => window.removeEventListener('mousemove', onMove)
   726→   726→  }, [])
   727→   727→
   728→   728→  useFrame(() => {
   729→   729→    if (!sceneVisible.current) return
   730→   730→
   731→   731→    const scrollY = window.scrollY
   732→   732→    const vh = window.innerHeight
   733→   733→    const sp = Math.min(1, scrollY / (vh * 2.5))
   734→   734→
   735→   735→    // Camera zooms out on scroll
   736→   736→    const z = THREE.MathUtils.lerp(7.5, 13, sp)
   737→   737→    // Mouse parallax reduces with scroll
   738→   738→    const parallaxStrength = 1 - sp * 0.6
   739→   739→    const x = mouse.current.x * 0.7 * parallaxStrength
   740→   740→    const y = -mouse.current.y * 0.4 * parallaxStrength
   741→   741→
   742→   742→    target.current.set(x, y, z)
   743→   743→    camera.position.lerp(target.current, 0.04)
   744→   744→    camera.lookAt(0, 0, 0)
   745→   745→  })
   746→   746→
   747→   747→  return null
   748→   748→}
